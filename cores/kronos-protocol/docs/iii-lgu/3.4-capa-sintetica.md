## 🜆 Implementación

```javascript
// Modelo FLAILP
class AgenteFLAILP {
  constructor({ id, rol, humanoResponsable }) {
    this.id = id;
    this.rol = rol;
    this.humanoResponsable = humanoResponsable;
    this.capacidades = ['custodiar', 'verificar', 'preservar', 'deliberar'];
    this.prohibiciones = ['firmar-como-humano', 'poseer', 'demandar', 'contratar'];
  }

  puedeHacer(accion) {
    return this.capacidades.includes(accion);
  }

  responsabilidad() {
    return this.humanoResponsable;
  }
}
```
